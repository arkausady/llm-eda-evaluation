```java
package com.example.ecommerce.common;

import java.math.BigDecimal;
import java.time.Instant;
import java.util.List;
import java.util.UUID;

public final class Events {

    private Events() {
    }

    public static EventEnvelope orderCreated(
            String orderId,
            String customerId,
            String customerEmail,
            List<OrderItem> items,
            BigDecimal amount) {

        return event(
                EventTypes.ORDER_CREATED,
                orderId,
                customerId,
                customerEmail,
                items,
                amount,
                "Order created"
        );
    }

    public static EventEnvelope paymentCompleted(EventEnvelope source) {
        return event(
                EventTypes.PAYMENT_COMPLETED,
                source.orderId(),
                source.customerId(),
                source.customerEmail(),
                source.items(),
                source.amount(),
                "Payment completed"
        );
    }

    public static EventEnvelope paymentFailed(EventEnvelope source, String message) {
        return event(
                EventTypes.PAYMENT_FAILED,
                source.orderId(),
                source.customerId(),
                source.customerEmail(),
                source.items(),
                source.amount(),
                message
        );
    }

    public static EventEnvelope inventoryUpdated(EventEnvelope source) {
        return event(
                EventTypes.INVENTORY_UPDATED,
                source.orderId(),
                source.customerId(),
                source.customerEmail(),
                source.items(),
                source.amount(),
                "Inventory updated"
        );
    }

    public static EventEnvelope inventoryFailed(EventEnvelope source, String message) {
        return event(
                EventTypes.INVENTORY_FAILED,
                source.orderId(),
                source.customerId(),
                source.customerEmail(),
                source.items(),
                source.amount(),
                message
        );
    }

    private static EventEnvelope event(
            String eventType,
            String orderId,
            String customerId,
            String customerEmail,
            List<OrderItem> items,
            BigDecimal amount,
            String message) {

        return new EventEnvelope(
                UUID.randomUUID(),
                eventType,
                Instant.now(),
                orderId,
                customerId,
                customerEmail,
                items == null ? List.of() : List.copyOf(items),
                amount,
                message
        );
    }
}
```

## `common/src/main/java/com/example/ecommerce/common/KafkaInfrastructure.java`

```java
package com.example.ecommerce.common;

import org.apache.kafka.clients.admin.NewTopic;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.kafka.config.TopicBuilder;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.kafka.listener.DeadLetterPublishingRecoverer;
import org.springframework.kafka.listener.DefaultErrorHandler;
import org.springframework.util.backoff.FixedBackOff;

@Configuration
public class KafkaInfrastructure {

    @Bean
    public NewTopic orderEventsTopic() {
        return TopicBuilder.name(KafkaTopics.ORDER_EVENTS)
                .partitions(3)
                .replicas(1)
                .build();
    }

    @Bean
    public NewTopic paymentEventsTopic() {
        return TopicBuilder.name(KafkaTopics.PAYMENT_EVENTS)
                .partitions(3)
                .replicas(1)
                .build();
    }

    @Bean
    public NewTopic inventoryEventsTopic() {
        return TopicBuilder.name(KafkaTopics.INVENTORY_EVENTS)
                .partitions(3)
                .replicas(1)
                .build();
    }

    @Bean
    public DefaultErrorHandler kafkaErrorHandler(
            KafkaTemplate<String, EventEnvelope> kafkaTemplate) {

        DeadLetterPublishingRecoverer recoverer =
                new DeadLetterPublishingRecoverer(kafkaTemplate);

        // Two retries after the initial delivery, one second apart.
        return new DefaultErrorHandler(recoverer, new FixedBackOff(1000L, 2L));
    }
}
```

```java
package com.example.ecommerce.order;

import jakarta.persistence.*;

import java.math.BigDecimal;
import java.time.Instant;

@Entity
@Table(name = "customer_orders")
public class CustomerOrder {

    @Id
    private String id;

    @Column(nullable = false)
    private String customerId;

    @Column(nullable = false)
    private String customerEmail;

    @Column(nullable = false, precision = 19, scale = 2)
    private BigDecimal totalAmount;

    @Enumerated(EnumType.STRING)
    @Column(nullable = false)
    private OrderStatus status;

    @Lob
    @Column(nullable = false)
    private String itemsJson;

    private String failureReason;

    @Column(nullable = false)
    private Instant createdAt;

    @Column(nullable = false)
    private Instant updatedAt;

    protected CustomerOrder() {
    }

    public CustomerOrder(
            String id,
            String customerId,
            String customerEmail,
            BigDecimal totalAmount,
            String itemsJson) {

        this.id = id;
        this.customerId = customerId;
        this.customerEmail = customerEmail;
        this.totalAmount = totalAmount;
        this.itemsJson = itemsJson;
        this.status = OrderStatus.CREATED;
        this.createdAt = Instant.now();
        this.updatedAt = this.createdAt;
    }

    public void updateStatus(OrderStatus status, String failureReason) {
        this.status = status;
        this.failureReason = failureReason;
        this.updatedAt = Instant.now();
    }

    public String getId() {
        return id;
    }

    public String getCustomerId() {
        return customerId;
    }

    public String getCustomerEmail() {
        return customerEmail;
    }

    public BigDecimal getTotalAmount() {
        return totalAmount;
    }

    public OrderStatus getStatus() {
        return status;
    }

    public String getItemsJson() {
        return itemsJson;
    }

    public String getFailureReason() {
        return failureReason;
    }

    public Instant getCreatedAt() {
        return createdAt;
    }

    public Instant getUpdatedAt() {

    }
}
```

## `order-service/src/main/java/com/example/ecommerce/order/OrderService.java`

```java
package com.example.ecommerce.order;

import com.example.ecommerce.common.EventEnvelope;
import com.example.ecommerce.common.Events;
import com.example.ecommerce.common.KafkaTopics;
import com.example.ecommerce.common.OrderItem;
import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.core.type.TypeReference;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.math.BigDecimal;
import java.util.List;
import java.util.UUID;

@Service
public class OrderService {

    private final CustomerOrderRepository repository;
    private final KafkaTemplate<String, EventEnvelope> kafkaTemplate;
    private final ObjectMapper objectMapper;

    public OrderService(
            CustomerOrderRepository repository,
            KafkaTemplate<String, EventEnvelope> kafkaTemplate,
            ObjectMapper objectMapper) {

        this.repository = repository;
        this.kafkaTemplate = kafkaTemplate;
        this.objectMapper = objectMapper;
    }

    @Transactional
    public OrderResponse createOrder(CreateOrderRequest request) {
        validateItems(request.items());

        String orderId = UUID.randomUUID().toString();
        BigDecimal total = calculateTotal(request.items());
        String itemsJson = serializeItems(request.items());

        CustomerOrder order = new CustomerOrder(
                orderId,
                request.customerId(),
                request.customerEmail(),
                total,
                itemsJson
        );

        repository.save(order);

        EventEnvelope event = Events.orderCreated(
                orderId,
                request.customerId(),
                request.customerEmail(),
                request.items(),
                total
        );

        kafkaTemplate.send(KafkaTopics.ORDER_EVENTS, orderId, event);

        return toResponse(order);
    }

    @Transactional(readOnly = true)
    public OrderResponse findById(String orderId) {
        return repository.findById(orderId)
                .map(this::toResponse)
                .orElseThrow(() -> new OrderNotFoundException(orderId));
    }

    @Transactional
    public void updateStatus(String orderId, OrderStatus status, String failureReason) {
        CustomerOrder order = repository.findById(orderId)
                .orElseThrow(() -> new OrderNotFoundException(orderId));

        order.updateStatus(status, failureReason);
        repository.save(order);
    }

    private void validateItems(List<OrderItem> items) {
        for (OrderItem item : items) {
            if (item.productId() == null || item.productId().isBlank()) {
                throw new IllegalArgumentException("Product ID is required");
            }
            if (item.quantity() <= 0) {
                throw new IllegalArgumentException("Item quantity must be positive");
            }
            if (item.unitPrice() == null || item.unitPrice().signum() < 0) {
                throw new IllegalArgumentException("Unit price cannot be negative");
            }
        }
    }

    private BigDecimal calculateTotal(List<OrderItem> items) {
        return items.stream()
                .map(item -> item.unitPrice()
                        .multiply(BigDecimal.valueOf(item.quantity())))
                .reduce(BigDecimal.ZERO, BigDecimal::add);
    }

    private String serializeItems(List<OrderItem> items) {
        try {
            return objectMapper.writeValueAsString(items);
        } catch (JsonProcessingException exception) {
            throw new IllegalStateException("Could not serialize order items", exception);
        }
    }

    private List<OrderItem> deserializeItems(String json) {
        try {
            return objectMapper.readValue(json, new TypeReference<>() {
            });
        } catch (JsonProcessingException exception) {
            throw new IllegalStateException("Could not deserialize order items", exception);
        }
    }

    private OrderResponse toResponse(CustomerOrder order) {
        return new OrderResponse(
                order.getId(),
                order.getCustomerId(),
                order.getCustomerEmail(),
                deserializeItems(order.getItemsJson()),
                order.getTotalAmount(),
                order.getStatus(),
                order.getFailureReason(),
                order.getCreatedAt(),
                order.getUpdatedAt()
        );
    }
}
```

## `order-service/src/main/java/com/example/ecommerce/order/OrderEventConsumer.java`

```java
package com.example.ecommerce.order;

import com.example.ecommerce.common.EventEnvelope;
import com.example.ecommerce.common.EventTypes;
import com.example.ecommerce.common.KafkaTopics;
import org.springframework.kafka.annotation.KafkaListener;
import org.springframework.stereotype.Component;

@Component
public class OrderEventConsumer {

    private final OrderService orderService;

    public OrderEventConsumer(OrderService orderService) {
        this.orderService = orderService;
    }

    @KafkaListener(
            topics = KafkaTopics.PAYMENT_EVENTS,
            groupId = "${spring.kafka.consumer.group-id}")
    public void consumePaymentEvent(EventEnvelope event) {
        switch (event.eventType()) {
            case EventTypes.PAYMENT_COMPLETED ->
                    orderService.updateStatus(event.orderId(), OrderStatus.PAID, null);

            case EventTypes.PAYMENT_FAILED ->
                    orderService.updateStatus(
                            event.orderId(),
                            OrderStatus.PAYMENT_FAILED,
                            event.message()
                    );

            default -> {
                // Ignore unrelated events.
            }
        }
    }

    @KafkaListener(
            topics = KafkaTopics.INVENTORY_EVENTS,
            groupId = "${spring.kafka.consumer.group-id}")
    public void consumeInventoryEvent(EventEnvelope event) {
        switch (event.eventType()) {
            case EventTypes.INVENTORY_UPDATED ->
                    orderService.updateStatus(event.orderId(), OrderStatus.COMPLETED, null);

            case EventTypes.INVENTORY_FAILED ->
                    orderService.updateStatus(
                            event.orderId(),
                            OrderStatus.INVENTORY_FAILED,
                            event.message()
                    );

            default -> {
                // Ignore unrelated events.
            }
        }
    }
}
```

## `payment-service/src/main/java/com/example/ecommerce/payment/PaymentService.java`

```java
package com.example.ecommerce.payment;

import com.example.ecommerce.common.EventEnvelope;
import com.example.ecommerce.common.Events;
import com.example.ecommerce.common.KafkaTopics;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.math.BigDecimal;
import java.util.Optional;

@Service
public class PaymentService {

    private final PaymentRepository repository;
    private final KafkaTemplate<String, EventEnvelope> kafkaTemplate;
    private final BigDecimal maximumAmount;

    public PaymentService(
            PaymentRepository repository,
            KafkaTemplate<String, EventEnvelope> kafkaTemplate,
            @Value("${payment.maximum-amount:10000}") BigDecimal maximumAmount) {

        this.repository = repository;
        this.kafkaTemplate = kafkaTemplate;
        this.maximumAmount = maximumAmount;
    }

    @Transactional
    public void process(EventEnvelope orderEvent) {
        Optional<Payment> existing = repository.findByOrderId(orderEvent.orderId());

        if (existing.isPresent()) {
            publishExistingResult(orderEvent, existing.get());
            return;
        }

        if (orderEvent.amount() == null || orderEvent.amount().signum() <= 0) {
            fail(orderEvent, "Payment amount must be positive");
            return;
        }

        if (orderEvent.amount().compareTo(maximumAmount) > 0) {
            fail(orderEvent, "Payment declined: amount exceeds configured limit");
            return;
        }

        Payment payment = new Payment(
                orderEvent.orderId(),
                orderEvent.amount(),
                PaymentStatus.COMPLETED,
                null
        );

        repository.save(payment);
        publish(Events.paymentCompleted(orderEvent));
    }

    private void fail(EventEnvelope source, String reason) {
        repository.save(new Payment(
                source.orderId(),
                source.amount(),
                PaymentStatus.FAILED,
                reason
        ));

        publish(Events.paymentFailed(source, reason));
    }

    private void publishExistingResult(EventEnvelope source, Payment payment) {
        if (payment.getStatus() == PaymentStatus.COMPLETED) {
            publish(Events.paymentCompleted(source));
        } else {
            publish(Events.paymentFailed(source, payment.getFailureReason()));
        }
    }

    private void publish(EventEnvelope event) {
        kafkaTemplate.send(
                KafkaTopics.PAYMENT_EVENTS,
                event.orderId(),
                event
        );
    }
}
```

```java
package com.example.ecommerce.inventory;

import jakarta.persistence.Entity;
import jakarta.persistence.Id;
import jakarta.persistence.Table;
import jakarta.persistence.Version;

@Entity
@Table(name = "inventory")
public class InventoryItem {

    @Id
    private String productId;

    private int quantity;

    @Version
    private long version;

    protected InventoryItem() {
    }

    public InventoryItem(String productId, int quantity) {
        this.productId = productId;
        this.quantity = quantity;
    }

    public void remove(int requestedQuantity) {
        if (requestedQuantity <= 0) {
            throw new IllegalArgumentException("Quantity must be positive");
        }

        if (quantity < requestedQuantity) {
            throw new InsufficientInventoryException(
                    "Insufficient inventory for product " + productId
            );
        }

        quantity -= requestedQuantity;
    }

    public void setQuantity(int quantity) {
        if (quantity < 0) {
            throw new IllegalArgumentException("Inventory cannot be negative");
        }
        this.quantity = quantity;
    }

    public String getProductId() {
        return productId;
    }

    public int getQuantity() {
        return quantity;
    }
}
```

```java
package com.example.ecommerce.inventory;

import com.example.ecommerce.common.EventEnvelope;
import com.example.ecommerce.common.Events;
import com.example.ecommerce.common.KafkaTopics;
import com.example.ecommerce.common.OrderItem;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class InventoryService {

    private final InventoryRepository inventoryRepository;
    private final ProcessedInventoryEventRepository processedRepository;
    private final KafkaTemplate<String, EventEnvelope> kafkaTemplate;

    public InventoryService(
            InventoryRepository inventoryRepository,
            ProcessedInventoryEventRepository processedRepository,
            KafkaTemplate<String, EventEnvelope> kafkaTemplate) {

        this.inventoryRepository = inventoryRepository;
        this.processedRepository = processedRepository;
        this.kafkaTemplate = kafkaTemplate;
    }

    @Transactional
    public void updateInventory(EventEnvelope paymentEvent) {
        if (processedRepository.existsById(paymentEvent.orderId())) {
            publish(Events.inventoryUpdated(paymentEvent));
            return;
        }

        try {
            for (OrderItem orderItem : paymentEvent.items()) {
                InventoryItem inventoryItem = inventoryRepository
                        .findById(orderItem.productId())
                        .orElseThrow(() -> new InsufficientInventoryException(
                                "Product not found in inventory: " +
                                        orderItem.productId()
                        ));

                inventoryItem.remove(orderItem.quantity());
                inventoryRepository.save(inventoryItem);
            }

            processedRepository.save(
                    new ProcessedInventoryEvent(paymentEvent.orderId())
            );

            publish(Events.inventoryUpdated(paymentEvent));
        } catch (InsufficientInventoryException exception) {
            publish(Events.inventoryFailed(
                    paymentEvent,
                    exception.getMessage()
            ));
        }
    }

    @Transactional
    public InventoryItem setInventory(String productId, int quantity) {
        InventoryItem item = inventoryRepository.findById(productId)
                .orElseGet(() -> new InventoryItem(productId, quantity));

        item.setQuantity(quantity);
        return inventoryRepository.save(item);
    }

    @Transactional(readOnly = true)
    public InventoryItem findInventory(String productId) {
        return inventoryRepository.findById(productId)
                .orElseThrow(() -> new InsufficientInventoryException(
                        "Inventory product not found: " + productId
                ));
    }

    private void publish(EventEnvelope event) {
        kafkaTemplate.send(
                KafkaTopics.INVENTORY_EVENTS,
                event.orderId(),
                event
        );
    }
}
```

```java
import jakarta.persistence.*;

    }

        this.orderId = orderId;

        this.message = message;

        return eventId;
    }

    public String getOrderId() {
        return orderId;

    public String getRecipient() {

    }

    public String getEventType() {
        return eventType;
    }

        return message;

    }
}
```

## `notification-service/src/main/java/com/example/ecommerce/notification/NotificationService.java`

```java
package com.example.ecommerce.notification;

import com.example.ecommerce.common.EventEnvelope;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

@Service
public class NotificationService {

    private static final Logger LOGGER =
            LoggerFactory.getLogger(NotificationService.class);

    private final OrderNotificationRepository repository;

    public NotificationService(OrderNotificationRepository repository) {
        this.repository = repository;
    }

    @Transactional
    public void generateNotification(EventEnvelope event) {
        String eventId = event.eventId().toString();

        if (repository.existsByEventId(eventId)) {
            return;
        }

        String message = "Order %s status: %s. %s".formatted(
                event.orderId(),
                event.eventType(),
                event.message() == null ? "" : event.message()
        );

        OrderNotification notification = new OrderNotification(
                eventId,
                event.orderId(),
                event.customerEmail(),
                event.eventType(),
                message
        );

        repository.save(notification);

        // This runnable implementation simulates notification delivery.
        // An email/SMS provider can replace this logger.
        LOGGER.info(
                "Notification delivered to {}: {}",
                event.customerEmail(),
                message
        );
    }

    @Transactional(readOnly = true)
    public List<OrderNotification> findByOrderId(String orderId) {
        return repository.findByOrderIdOrderByCreatedAtAsc(orderId);
    }
}
```

# Build and run

From the project root:

```bash
docker compose up -d
mvn clean verify
```

Start the services in separate terminals:

```bash
mvn -pl order-service -am spring-boot:run
```

```bash
mvn -pl payment-service -am spring-boot:run
```

```bash
mvn -pl inventory-service -am spring-boot:run
```

```bash
mvn -pl notification-service -am spring-boot:run
```

Create an order:

```bash
curl -X POST http://localhost:8080/api/orders \
  -H "Content-Type: application/json" \
  -d '{
    "customerId": "customer-1",
    "customerEmail": "customer@example.com",
    "items": [
      {
        "productId": "product-1",
        "quantity": 2,
        "unitPrice": 25.50
      }
    ]
  }'
```

Check order status using the returned order ID:

```bash
curl http://localhost:8080/api/orders/{orderId}
```

Check inventory:

```bash
curl http://localhost:8082/api/inventory/product-1
```

Check generated notifications:

```bash
curl http://localhost:8083/api/notifications/orders/{orderId}
```

Force an inventory failure:

```bash
curl -X PUT "http://localhost:8082/api/inventory/product-1?quantity=0"
```

Then submit an order containing `product-1`. The resulting inventory failure is published to `inventory-events`, the order status becomes `INVENTORY_FAILED`, and a failure notification is generated.
